# Diretórios contendo regras YARA em formato fonte (.yar, .yara) — o YARA-X
# compila o catálogo em runtime no startup do caso. Formatos pré-compilados
# (.yarc do YARA clássico ou serialização própria do YARA-X) NÃO são aceitos
# na v1; ver specs/001-yara-rules-engine/research.md §R-01.
#
# Múltiplos caminhos podem ser separados por ";" (Windows) ou ":" (Linux).
# Quando vazio, a feature fica efetivamente desligada para o caso.
#
# Formas suportadas:
#   - Caminho absoluto:     C:\rules\yara          ou  /opt/iped/rules
#   - ${IPED_HOME} token:   ${IPED_HOME}/yara-rules   (recomendado — expande
#                           para a raiz da instalação do IPED em runtime)
#   - Caminho relativo:     yara-rules            (resolvido a partir do diretório
#                           de trabalho do JVM; FUNCIONA quando o IPED é iniciado
#                           pela sua pasta de instalação, frágil em outros casos
#                           — prefira ${IPED_HOME}/...)
ruleDirectories =

# Limite máximo de tamanho por item a ser escaneado.
# Aceita sufixos K, M, G (case-insensitive, base 1024). Itens acima do
# limite são pulados e o motivo é registrado.
maxFileSizeBytes = 250M

# Limite máximo de tempo por item, em milissegundos. Scans que excedam
# o limite são interrompidos sem propagar exceção; o item é contado
# como "skipped" nas métricas.
perItemTimeoutMs = 30000

# Quando false (default), apenas itens com content stream binário
# acessível (arquivos, subitens de containers, carved items) são
# escaneados. Quando true, força tentativa em todos os itens — útil
# para auditoria; tem custo de performance maior.
scanAllItems = false

# Tamanho máximo (em bytes) dos trechos brutos persistidos no JSON de
# detalhe de match. Excesso é truncado e marcado com "truncated": true.
# Faixa válida: 1 a 65536.
matchHexMaxBytes = 256
